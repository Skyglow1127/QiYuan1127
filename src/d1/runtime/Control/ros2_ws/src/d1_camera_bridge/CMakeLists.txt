cmake_minimum_required(VERSION 3.16)
project(d1_camera_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED)
set(CAMERA_SDK_DIR "/home/aup/Camera/SDK" CACHE PATH "Read-only existing Orbbec SDK")
if(NOT EXISTS "${CAMERA_SDK_DIR}/lib/libOrbbecSDK.so")
  message(FATAL_ERROR "Set -DCAMERA_SDK_DIR to Camera/SDK. No SDK files will be modified.")
endif()
add_executable(camera_bridge src/camera_bridge.cpp)
target_include_directories(camera_bridge PRIVATE ${CAMERA_SDK_DIR}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_bridge "${CAMERA_SDK_DIR}/lib/libOrbbecSDK.so" ${OpenCV_LIBS})
ament_target_dependencies(camera_bridge rclcpp sensor_msgs std_msgs cv_bridge)
target_compile_options(camera_bridge PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(camera_bridge PROPERTIES BUILD_RPATH "${CAMERA_SDK_DIR}/lib" INSTALL_RPATH "${CAMERA_SDK_DIR}/lib")
install(TARGETS camera_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
